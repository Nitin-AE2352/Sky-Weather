def _request(url, params, headers=None):
    try:
        response = httpx.get(
            url,
            params=params,
            headers=headers,
            timeout=TIMEOUT,
            follow_redirects=True
        )

        response.raise_for_status()
        return response.json()

    except httpx.TimeoutException:
        raise Exception("Weather API request timed out.")

    except httpx.HTTPStatusError as error:
        raise Exception(
            f"Weather API returned HTTP {error.response.status_code}."
        )

    except httpx.RequestError:
        raise Exception("Unable to connect to weather service.")

    except ValueError:
        raise Exception("Weather API returned invalid JSON.")


def search_cities(city_name, count=10):
    city_name = city_name.strip()

    if not city_name:
        return []

    data = _request(
        GEOCODING_URL,
        {
            "name": city_name,
            "count": count,
            "language": "en",
            "format": "json"
        }
    )

    results = []

    for city in data.get("results", []):
        results.append({
            "id": city.get("id"),
            "name": city.get("name"),
            "country": city.get("country"),
            "country_code": city.get("country_code"),
            "admin1": city.get("admin1"),
            "admin2": city.get("admin2"),
            "latitude": city.get("latitude"),
            "longitude": city.get("longitude"),
            "timezone": city.get("timezone"),
            "population": city.get("population")
        })

    return results


def search_city(city_name):
    cities = search_cities(city_name, count=10)

    if not cities:
        raise Exception(f"City '{city_name}' not found.")

    exact_matches = [
        city for city in cities
        if city["name"].lower() == city_name.strip().lower()
    ]

    if exact_matches:
        cities = exact_matches

    india_matches = [
        city for city in cities
        if city.get("country_code") == "IN"
    ]

    if len(india_matches) == 1:
        return india_matches[0]

    if len(cities) == 1:
        return cities[0]

    return cities[0]


def _validate_coordinates(latitude, longitude):
    if latitude is None or longitude is None:
        raise ValueError("Latitude and longitude are required.")

    if not -90 <= float(latitude) <= 90:
        raise ValueError("Invalid latitude.")

    if not -180 <= float(longitude) <= 180:
        raise ValueError("Invalid longitude.")


def _get_open_meteo_weather(latitude, longitude, timezone_name):
    params = {
        "latitude": latitude,
        "longitude": longitude,
        "current": ",".join(CURRENT_FIELDS),
        "hourly": ",".join(HOURLY_FIELDS),
        "daily": ",".join(DAILY_FIELDS),
        "timezone": timezone_name,
        "forecast_days": 7,
        "past_days": 0,
        "temperature_unit": "celsius",
        "wind_speed_unit": "kmh",
        "precipitation_unit": "mm"
    }

    data = _request(WEATHER_URL, params)

    current = data.get("current")

    if not current:
        raise Exception("Open-Meteo current data unavailable.")

    current["description"] = weather_description(
        current.get("weather_code")
    )

    daily = data.get("daily", {})

    daily["description"] = [
        weather_description(code)
        for code in daily.get("weather_code", [])
    ]

    return {
        "current": current,
        "hourly": data.get("hourly", {}),
        "daily": daily,
        "timezone": data.get("timezone"),
        "timezone_abbreviation": data.get("timezone_abbreviation"),
        "latitude": data.get("latitude"),
        "longitude": data.get("longitude"),
        "elevation": data.get("elevation"),
        "units": {
            "temperature": "°C",
            "wind_speed": "km/h",
            "precipitation": "mm",
            "pressure": "hPa"
        },
        "source": "Open-Meteo",
        "updated_at": current.get("time"),
        "fetched_at": datetime.now(timezone.utc).isoformat()
    }


def _met_symbol_description(symbol):
    if not symbol:
        return "Weather information unavailable"

    symbol = symbol.lower()

    if "thunder" in symbol:
        return "Thunderstorm"

    if "snow" in symbol:
        return "Snow"

    if "sleet" in symbol:
        return "Sleet"

    if "rain" in symbol:
        return "Rain"

    if "fog" in symbol:
        return "Fog"

    if "cloudy" in symbol:
        return "Cloudy"

    if "partlycloudy" in symbol:
        return "Partly cloudy"

    if "clearsky" in symbol:
        return "Clear sky"

    return "Weather conditions available"


def _get_met_weather(latitude, longitude, timezone_name):
    params = {
        "lat": latitude,
        "lon": longitude
    }

    data = _request(
        MET_URL,
        params,
        headers=HEADERS
    )

    series = data.get("properties", {}).get("timeseries", [])

    if not series:
        raise Exception("MET Norway forecast unavailable.")

    try:
        local_zone = ZoneInfo(timezone_name)
    except Exception:
        local_zone = timezone.utc

    now = datetime.now(timezone.utc)

    def parse_time(item):
        return datetime.fromisoformat(
            item["time"].replace("Z", "+00:00")
        )

    selected = min(
        series,
        key=lambda item: abs((parse_time(item) - now).total_seconds())
    )

    details = selected["data"]["instant"]["details"]

    next_data = selected["data"].get("next_1_hours", {})
    next_details = next_data.get("details", {})
    next_summary = next_data.get("summary", {})

    symbol = next_summary.get("symbol_code")

    wind_ms = details.get("wind_speed")
    gust_ms = details.get("wind_speed_of_gust")

    current_time = parse_time(selected).astimezone(
        local_zone
    )

    current = {
        "time": current_time.isoformat(),
        "temperature_2m": details.get("air_temperature"),
        "relative_humidity_2m": details.get("relative_humidity"),
        "apparent_temperature": None,
        "is_day": int(6 <= current_time.hour < 18),
        "precipitation": next_details.get("precipitation_amount", 0),
        "rain": next_details.get("precipitation_amount", 0),
        "showers": None,
        "snowfall": None,
        "weather_code": None,
        "cloud_cover": details.get("cloud_area_fraction"),
        "pressure_msl": details.get("air_pressure_at_sea_level"),
        "surface_pressure": None,
        "wind_speed_10m": (
            round(wind_ms * 3.6, 1)
            if wind_ms is not None else None
        ),
        "wind_direction_10m": details.get("wind_from_direction"),
        "wind_gusts_10m": (
            round(gust_ms * 3.6, 1)
            if gust_ms is not None else None
        ),
        "description": _met_symbol_description(symbol)
    }

    hourly = {
        "time": [],
        "temperature_2m": [],
        "relative_humidity_2m": [],
        "precipitation_probability": [],
        "precipitation": [],
        "rain": [],
        "weather_code": [],
        "cloud_cover": [],
        "wind_speed_10m": [],
        "wind_direction_10m": [],
        "uv_index": []
    }

    daily_groups = {}

    for item in series:
        item_time = parse_time(item).astimezone(local_zone)
        instant = item["data"]["instant"]["details"]

        hour_data = item["data"].get("next_1_hours", {})
        hour_details = hour_data.get("details", {})
        hour_summary = hour_data.get("summary", {})

        if item_time >= current_time and len(hourly["time"]) < 48:
            hourly["time"].append(item_time.isoformat())
            hourly["temperature_2m"].append(
                instant.get("air_temperature")
            )
            hourly["relative_humidity_2m"].append(
                instant.get("relative_humidity")
            )
            hourly["precipitation_probability"].append(None)

            rainfall = hour_details.get("precipitation_amount")

            hourly["precipitation"].append(rainfall)
            hourly["rain"].append(rainfall)
            hourly["weather_code"].append(None)
            hourly["cloud_cover"].append(
                instant.get("cloud_area_fraction")
            )

            wind = instant.get("wind_speed")

            hourly["wind_speed_10m"].append(
                round(wind * 3.6, 1)
                if wind is not None else None
            )

            hourly["wind_direction_10m"].append(
                instant.get("wind_from_direction")
            )

            hourly["uv_index"].append(None)

        date_key = item_time.date().isoformat()

        group = daily_groups.setdefault(
            date_key,
            {
                "temperatures": [],
                "rainfall": [],
                "winds": [],
                "symbols": []
            }
        )

        temperature = instant.get("air_temperature")

        if temperature is not None:
            group["temperatures"].append(temperature)

        rainfall = hour_details.get("precipitation_amount")

        if rainfall is not None:
            group["rainfall"].append(rainfall)

        wind = instant.get("wind_speed")

        if wind is not None:
            group["winds"].append(wind * 3.6)

        daily_symbol = hour_summary.get("symbol_code")

        if daily_symbol:
            group["symbols"].append(daily_symbol)

    daily = {
        "time": [],
        "weather_code": [],
        "temperature_2m_max": [],
        "temperature_2m_min": [],
        "apparent_temperature_max": [],
        "apparent_temperature_min": [],
        "sunrise": [],
        "sunset": [],
        "uv_index_max": [],
        "precipitation_sum": [],
        "precipitation_probability_max": [],
        "wind_speed_10m_max": [],
        "description": []
    }

    for date_key, group in sorted(daily_groups.items()):
        temperatures = group["temperatures"]
        rainfall = group["rainfall"]
        winds = group["winds"]

        daily["time"].append(date_key)
        daily["weather_code"].append(None)

        daily["temperature_2m_max"].append(
            max(temperatures) if temperatures else None
        )

        daily["temperature_2m_min"].append(
            min(temperatures) if temperatures else None
        )

        daily["apparent_temperature_max"].append(None)
        daily["apparent_temperature_min"].append(None)
        daily["sunrise"].append(None)
        daily["sunset"].append(None)
        daily["uv_index_max"].append(None)

        daily["precipitation_sum"].append(
            round(sum(rainfall), 2) if rainfall else None
        )

        daily["precipitation_probability_max"].append(None)

        daily["wind_speed_10m_max"].append(
            round(max(winds), 1) if winds else None
        )

        daily["description"].append(
            _met_symbol_description(
                group["symbols"][0] if group["symbols"] else None
            )
        )

    return {
        "current": current,
        "hourly": hourly,
        "daily": daily,
        "timezone": timezone_name,
        "timezone_abbreviation": None,
        "latitude": latitude,
        "longitude": longitude,
        "elevation": None,
        "units": {
            "temperature": "°C",
            "wind_speed": "km/h",
            "precipitation": "mm",
            "pressure": "hPa"
        },
        "source": "MET Norway",
        "updated_at": current["time"],
        "fetched_at": datetime.now(timezone.utc).isoformat()
    }


def get_weather(latitude, longitude, timezone_name="auto"):
    _validate_coordinates(latitude, longitude)

    try:
        return _get_open_meteo_weather(
            latitude,
            longitude,
            timezone_name
        )

    except Exception as primary_error:
        print(
            "Open-Meteo failed. Trying MET Norway:",
            primary_error
        )

        try:
            return _get_met_weather(
                latitude,
                longitude,
                timezone_name
            )

        except Exception as fallback_error:
            raise Exception(
                "Both weather providers failed. "
                f"Open-Meteo: {primary_error}. "
                f"MET Norway: {fallback_error}"
            )

def get_weather_bundle(city_name):
    city = search_city(city_name)

    weather = get_weather(
        city["latitude"],
        city["longitude"],
        city.get("timezone") or "auto"
    )

    try:
        air_quality = get_air_quality(
            city["latitude"],
            city["longitude"],
            city.get("timezone") or "auto"
        )

    except Exception as error:
        air_quality = {
            "error": str(error),
            "source": "Open-Meteo Air Quality"
        }

    weather["air_quality"] = air_quality

    return {
        "city": city,
        "weather": weather
    }

import os
from pathlib import Path

import ollama
from dotenv import load_dotenv


BASE_DIR = Path(__file__).resolve().parent.parent

load_dotenv(BASE_DIR / ".env")


OLLAMA_HOST = os.getenv(
    "OLLAMA_HOST",
    "http://localhost:11434"
)

OLLAMA_MODEL = os.getenv(
    "OLLAMA_MODEL",
    "qwen2.5:1.5b"
)


client = ollama.Client(
    host=OLLAMA_HOST,
    timeout=180
)


def check_ollama_connection():
    """Check Ollama service and installed models."""

    try:
        result = client.list()

        models = [
            model.model
            for model in result.models
        ]

        model_available = any(
            model.split(":")[0] == OLLAMA_MODEL.split(":")[0]
            for model in models
        )

        return {
            "connected": True,
            "model": OLLAMA_MODEL,
            "model_available": model_available,
            "available_models": models
        }

    except Exception as error:
        return {
            "connected": False,
            "model": OLLAMA_MODEL,
            "model_available": False,
            "error": str(error)
        }


def ask_weather_ai(question, weather_data):
    """Answer weather questions using supplied weather data."""

    if not question or not question.strip():
        return "Please enter a question."

    if not isinstance(weather_data, dict):
        return "Weather information is not available."

    prompt = f"""
You are an accurate AI Weather Assistant.

IMPORTANT RULES:

1. Use only the supplied weather information.
2. Never invent temperature, humidity, rainfall or wind speed.
3. If a value is missing, say it is unavailable.
4. Clearly distinguish current weather and forecast.
5. Explain weather information in simple language.
6. Reply in the same language as the user's question.
7. Use Celsius for temperature and km/h for wind.
8. Mention the location when possible.
9. Do not claim that model-based data is a live sensor reading.
10. Give concise and useful answers.

WEATHER INFORMATION:

{weather_data}

USER QUESTION:

{question}
"""

    try:

from concurrent.futures import ThreadPoolExecutor
from Backend.weather_api import (
    search_city,
    _get_open_meteo_weather,
    _get_met_weather,
)


def _safe_float(value):
    if value is None:
        return None

    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def _difference(first, second):
    first = _safe_float(first)
    second = _safe_float(second)

    if first is None or second is None:
        return None

    return round(abs(first - second), 2)


def _get_provider_weather(provider, latitude, longitude, timezone_name):
    try:
        if provider == "Open-Meteo":
            return _get_open_meteo_weather(
                latitude,
                longitude,
                timezone_name
            )

        if provider == "MET Norway":
            return _get_met_weather(
                latitude,
                longitude,
                timezone_name
            )

        raise ValueError("Unknown weather provider")

    except Exception as error:
        return {
            "error": str(error),
            "source": provider
        }


def compare_weather(city_name):
    city = search_city(city_name)

    latitude = city["latitude"]
    longitude = city["longitude"]
    timezone_name = city.get("timezone") or "auto"

    # Fetch both providers simultaneously.
    with ThreadPoolExecutor(max_workers=2) as executor:
        open_meteo_future = executor.submit(
            _get_provider_weather,
            "Open-Meteo",
            latitude,
            longitude,
            timezone_name
        )

        met_future = executor.submit(
            _get_provider_weather,
            "MET Norway",
            latitude,
            longitude,
            timezone_name
        )

        open_meteo = open_meteo_future.result()
        met_norway = met_future.result()

    providers = {
        "open_meteo": open_meteo,
        "met_norway": met_norway
    }

    available = [
        (name, data)
        for name, data in providers.items()
        if "error" not in data
    ]

    if not available:
        return {
            "success": False,
            "city": city,
            "message": "Both weather providers failed.",
            "providers": providers
        }

    if len(available) == 1:
        name, data = available[0]

        return {
            "success": True,
            "city": city,
            "comparison_available": False,
            "message": "Only one weather provider is available.",
            "selected_provider": data["source"],
            "weather": data,
            "providers": providers
        }

    first = open_meteo["current"]
    second = met_norway["current"]

    temperature_difference = _difference(
        first.get("temperature_2m"),
        second.get("temperature_2m")
    )

    wind_difference = _difference(
        first.get("wind_speed_10m"),
        second.get("wind_speed_10m")
    )

    humidity_difference = _difference(
        first.get("relative_humidity_2m"),
        second.get("relative_humidity_2m")
    )

    # Agreement thresholds are comparison rules,
    # not guarantees of weather accuracy.
    temperature_agreement = (
        temperature_difference is not None
        and temperature_difference <= 2
    )

    wind_agreement = (
        wind_difference is not None
        and wind_difference <= 10
    )

    humidity_agreement = (
        humidity_difference is not None
        and humidity_difference <= 10
    )

    comparable_checks = [
        value
        for value in [
            temperature_agreement,
            wind_agreement,
            humidity_agreement
        ]
        if value is not None
    ]

    agreement_count = sum(comparable_checks)

    if agreement_count == len(comparable_checks):
        agreement_status = "Good agreement"
    elif agreement_count > 0:
        agreement_status = "Partial agreement"
    else:
        agreement_status = "Significant disagreement"

    return {
        "success": True,
        "city": city,
        "comparison_available": True,
        "comparison": {
            "temperature_difference_c": temperature_difference,
            "wind_speed_difference_kmh": wind_difference,
            "humidity_difference_percent": humidity_difference,
            "temperature_agreement": temperature_agreement,
            "wind_agreement": wind_agreement,
            "humidity_agreement": humidity_agreement,
            "agreement_status": agreement_status,
            "checks_passed": agreement_count,
            "total_checks": len(comparable_checks)
        },
        "providers": providers,
        "note": (
            "Provider agreement does not guarantee actual accuracy. "
            "These are model-based weather estimates."
        )
    }

import sqlite3

# =====================================

BASE_DIR = Path(__file__).resolve().parent.parent

DATABASE_DIR = BASE_DIR / "Database"
DATABASE_DIR.mkdir(parents=True, exist_ok=True)

DB_PATH = DATABASE_DIR / "weather_history.db"


# =====================================
# DATABASE CONNECTION
# =====================================

def get_connection():
    connection = sqlite3.connect(
        DB_PATH,
        timeout=15
    )

    connection.row_factory = sqlite3.Row

    return connection


# =====================================
# CREATE DATABASE TABLE
# =====================================

def initialize_database():

    with get_connection() as connection:

        connection.execute("""
            CREATE TABLE IF NOT EXISTS weather_history (

                id INTEGER PRIMARY KEY AUTOINCREMENT,

                city_name TEXT NOT NULL,

                country TEXT,

                latitude REAL,

                longitude REAL,

                temperature REAL,

                feels_like REAL,

                humidity REAL,

                wind_speed REAL,

                pressure REAL,

                precipitation REAL,

                cloud_cover REAL,

                weather_condition TEXT,

                weather_code INTEGER,

                timezone TEXT,

                source TEXT,

                weather_data TEXT,

                recorded_at TEXT NOT NULL
            )
        """)

        connection.execute("""
            CREATE INDEX IF NOT EXISTS
            idx_weather_city
            ON weather_history(city_name)
        """)

        connection.execute("""
            CREATE INDEX IF NOT EXISTS
            idx_weather_recorded_at
            ON weather_history(recorded_at)
        """)


# =====================================
# SAVE WEATHER HISTORY
# =====================================

def save_weather_history(city, weather):

    try:

        current = weather["weather"]["current"]

        weather_json = json.dumps(
            weather,
            ensure_ascii=False
        )

        recorded_at = datetime.now(
            timezone.utc
        ).isoformat()

        with get_connection() as connection:

            cursor = connection.execute("""

                INSERT INTO weather_history (

                    city_name,
                    country,
                    latitude,
                    longitude,
                    temperature,
                    feels_like,
                    humidity,
                    wind_speed,
                    pressure,
                    precipitation,
                    cloud_cover,
                    weather_condition,
                    weather_code,
                    timezone,
                    source,
                    weather_data,
                    recorded_at

                )

                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)

            """, (

                city.get("name"),

                city.get("country"),

                city.get("latitude"),

                city.get("longitude"),

                current.get("temperature_2m"),

                current.get("apparent_temperature"),

                current.get("relative_humidity_2m"),

                current.get("wind_speed_10m"),

                current.get("pressure_msl"),

                current.get("precipitation"),

                current.get("cloud_cover"),

                current.get("description"),

                current.get("weather_code"),

                weather["weather"].get("timezone"),

                weather["weather"].get("source"),

                weather_json,

                recorded_at

            ))

            record_id = cursor.lastrowid

        return {
            "success": True,
            "record_id": record_id
        }

    except Exception as error:

        return {
            "success": False,
            "error": str(error)
        }


# =====================================

def get_weather_history(
    city_name=None,
    limit=50
):

    limit = max(
        1,
        min(int(limit), 500)
    )

    with get_connection() as connection:

        if city_name:

            rows = connection.execute("""

                SELECT *
                FROM weather_history

                WHERE LOWER(city_name) = LOWER(?)

                ORDER BY id DESC

                LIMIT ?

            """, (

                city_name,
                limit

            )).fetchall()

        else:

            rows = connection.execute("""

                SELECT *
                FROM weather_history

                ORDER BY id DESC

                LIMIT ?

            """, (

                limit,

            )).fetchall()

    return [
        dict(row)
        for row in rows
    ]

def get_weather_history_by_id(record_id):
